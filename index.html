from qsong.ionq_adapter import run_bell_circuit_ionq
from qsong.harmony import TWINKLE, events_from_tune, settings

events = events_from_tune(TWINKLE)
alphas, betas = settings(events)

# Submit note 1 to IonQ Aria-1 QPU
counts = run_bell_circuit_ionq(
    alpha=alphas[0],
    beta=betas[0],
    shots=1000,
    backend_name="ionq_qpu.aria-1"
)
print("IonQ Aria-1 Counts:", counts)
